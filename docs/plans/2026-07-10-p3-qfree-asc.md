# P3 — Q-Free ASC Adapter (dual-transport scaffold) Implementation Plan


**Goal:** Onboard the first real vendor — a `qfree-asc` `SnapshotReader` adapter that reuses the NTCIP base (standard SNMP status decode) and adds a typed, no-op-by-default seam for Q-Free's HTTP API — proving the "vendor = one folder, dual-transport, on the live adapter path" workflow end to end.

**Architecture:** `qfree-asc` parses a nested `connection.snmp` sub-block, builds the SNMP driver via `snmp.NewFactory(translatorReg)` (device_kind `asc`) exactly as `ntcip` does — reusing the `signalcontrol` translator as the NTCIP base. Its `Read()` runs the SNMP poll, then calls an `APIAugmenter` interface whose production default is a no-op; the real Q-Free API decoder plugs into that seam later. `connection` stays adapter-parsed (the core treats it opaquely), so `ntcip` keeps its flat shape and `qfree` uses `connection.{snmp,api}` with no core change. This is P3 of the vendor-adapter design (`docs/specs/2026-07-10-vendor-adapter-architecture-design.md`).

**Tech Stack:** Go 1.26. Reuses `internal/adapter`, `sdk/drivers/snmp`, `internal/translator`, `internal/snmp`, `sdk/driver`.

## Scope boundaries

- Does **not** invent Q-Free enterprise OIDs or the HTTP API schema (we don't have them). The API path is a typed `APIAugmenter` seam with a no-op default — no fabricated Q-Free semantics ship.
- Does **not** add the `EventProducer` capability — the API "augments the same Reading" per the design decision, so `qfree-asc` is a single-capability `SnapshotReader`.
- Reuses the `signalcontrol` translator as the NTCIP base rather than duplicating OID logic.

## Global Constraints

- Module path `github.com/Vikasa2M/openits-collector`. Go 1.26. `go test ./...` resolves `openits-models` via `replace => ../openits-models`.
- Registry key is `qfree-asc` (`vendor` = `qfree`, `device_kind` = `asc`).
- `connection` for qfree is nested: `connection.snmp` (SNMP params, same flat keys the snmp driver reads) and optional `connection.api`. yaml.v3 decodes nested mappings under a `map[string]any` value as `map[string]any`.
- Behavior-preserving for everything else: purely additive packages + one `main` wiring line. Full suite (12 packages) stays green; the binary must still build + link.
- Commit after each task; `gofmt -s -w .` before committing.

```go
// Package qfree provides Q-Free vendor adapters. qfree-asc is a dual-transport
// SnapshotReader: it reads controller status over standard NTCIP SNMP (reusing
// the shared SNMP driver + signalcontrol translator as the NTCIP base) and has
// a typed seam (APIAugmenter) for Q-Free's HTTP API, whose production default
// is a no-op until the vendor API schema is wired.
package qfree

import (
	"context"
	"fmt"

	"github.com/Vikasa2M/openits-collector/internal/adapter"
	internalsnmp "github.com/Vikasa2M/openits-collector/internal/snmp"
	"github.com/Vikasa2M/openits-collector/internal/translator"
	"github.com/Vikasa2M/openits-collector/sdk/driver"
	"github.com/Vikasa2M/openits-collector/sdk/drivers/snmp"
)

// Vendor is the vendor token for Q-Free adapters.
const Vendor = "qfree"

// APIAugmenter enriches a Reading with data from Q-Free's HTTP API. The
// production default (noopAugmenter) does nothing; the real Q-Free API client
// implements this when the vendor API schema is available. Augmentation is
// best-effort — a failure must not discard the SNMP snapshot.
type APIAugmenter interface {
	Augment(ctx context.Context, r *driver.Reading) error
}

type noopAugmenter struct{}

func (noopAugmenter) Augment(context.Context, *driver.Reading) error { return nil }

// ascAdapter is the qfree-asc SnapshotReader: SNMP snapshot + API augment seam.
type ascAdapter struct {
	desc adapter.Descriptor
	drv  driver.Driver
	aug  APIAugmenter
}

var _ adapter.SnapshotReader = (*ascAdapter)(nil)

func (a *ascAdapter) Descriptor() adapter.Descriptor { return a.desc }

// Read polls SNMP for the standard NTCIP status, then applies the API
// augmenter. An augmenter error is logged-by-omission (best-effort): the SNMP
// snapshot is still returned so a Q-Free API outage doesn't blind the poller.
func (a *ascAdapter) Read(ctx context.Context) (*driver.Reading, error) {
	r, err := a.drv.Collect(ctx)
	if err != nil {
		return nil, err
	}
	_ = a.aug.Augment(ctx, r)
	return r, nil
}

func (a *ascAdapter) Close() error { return a.drv.Close() }

// SNMPClient exposes the wrapped driver's SNMP client for the command path.
func (a *ascAdapter) SNMPClient() internalsnmp.SNMPClient {
	if sc, ok := a.drv.(interface {
		SNMPClient() internalsnmp.SNMPClient
	}); ok {
		return sc.SNMPClient()
	}
	return nil
}

// newASCAdapter wraps a driver + augmenter. Exposed unexported for tests to
// inject a fake driver and augmenter without SNMP or a real API.
func newASCAdapter(desc adapter.Descriptor, drv driver.Driver, aug APIAugmenter) *ascAdapter {
	return &ascAdapter{desc: desc, drv: drv, aug: aug}
}

// subMap extracts a nested map[string]any sub-block from a connection map.
func subMap(cfg map[string]any, key string) (map[string]any, bool) {
	if cfg == nil {
		return nil, false
	}
	m, ok := cfg[key].(map[string]any)
	return m, ok
}

// ascFactory builds a qfree-asc adapter from the connection block. It requires a
// connection.snmp sub-block (the SNMP transport); connection.api is parsed by a
// future augmenter and is optional today (defaults to the no-op augmenter).
func ascFactory(translatorReg *translator.TranslatorRegistry) adapter.Factory {
	snmpFactory := snmp.NewFactory(translatorReg)
	return func(deviceID string, cfg map[string]any) (adapter.Adapter, error) {
		snmpCfg, ok := subMap(cfg, "snmp")
		if !ok {
			return nil, fmt.Errorf("qfree-asc %s: connection.snmp block is required", deviceID)
		}
		dc := driver.Config{}
		for k, v := range snmpCfg {
			dc[k] = v
		}
		dc[snmp.CfgDeviceKind] = "asc"
		dc[snmp.CfgDeviceID] = deviceID
		drv, err := snmpFactory(dc)
		if err != nil {
			return nil, err
		}
		// Seam: when connection.api is present and the Q-Free API schema is
		// wired, construct the real augmenter here. Until then, no-op.
		var aug APIAugmenter = noopAugmenter{}
		desc := adapter.Descriptor{Vendor: Vendor, DeviceKind: "asc", Caps: adapter.CapSnapshot}
		return newASCAdapter(desc, drv, aug), nil
	}
}

// RegisterTo registers qfree-asc with the adapter registry.
func RegisterTo(adapterReg *adapter.Registry, translatorReg *translator.TranslatorRegistry) {
	adapterReg.Register(Vendor, "asc", adapter.CapSnapshot, ascFactory(translatorReg))
}
```

```go
package qfree

import (
	"context"
	"testing"
	"time"

	"github.com/Vikasa2M/openits-collector/internal/adapter"
	internalsnmp "github.com/Vikasa2M/openits-collector/internal/snmp"
	"github.com/Vikasa2M/openits-collector/internal/translator"
	"github.com/Vikasa2M/openits-collector/sdk/driver"
)

type fakeDriver struct{ reading *driver.Reading }

func (f *fakeDriver) Collect(context.Context) (*driver.Reading, error) { return f.reading, nil }
func (f *fakeDriver) Close() error                                     { return nil }

type fakeAugmenter struct{ called bool }

func (f *fakeAugmenter) Augment(_ context.Context, r *driver.Reading) error {
	f.called = true
	r.ActivePlanID = 99 // prove the augment point ran and can mutate the Reading
	return nil
}

// Read must poll SNMP (Collect) THEN run the augmenter, merging into the Reading.
func TestASCAdapter_ReadPollsThenAugments(t *testing.T) {
	base := &driver.Reading{ControllerID: "asc-100", SampledAt: time.Unix(1, 0), ActivePlanID: 3}
	aug := &fakeAugmenter{}
	a := newASCAdapter(
		adapter.Descriptor{Vendor: Vendor, DeviceKind: "asc", Caps: adapter.CapSnapshot},
		&fakeDriver{reading: base},
		aug,
	)

	got, err := a.Read(context.Background())
	if err != nil {
		t.Fatalf("Read: %v", err)
	}
	if !aug.called {
		t.Error("augmenter was not called")
	}
	if got.ActivePlanID != 99 {
		t.Errorf("augmenter did not merge into Reading: ActivePlanID = %d, want 99", got.ActivePlanID)
	}
	if a.Descriptor().Key() != "qfree-asc" {
		t.Errorf("Key = %q, want qfree-asc", a.Descriptor().Key())
	}
}

// The registered factory builds from a nested connection.snmp block, is a
// network-free SnapshotReader (stub translator), and exposes SNMPClient().
func TestRegisterTo_BuildsFromNestedConnection(t *testing.T) {
	adapterReg := adapter.NewRegistry()
	transReg := translator.NewTranslatorRegistry()
	translator.RegisterStubs(transReg)
	RegisterTo(adapterReg, transReg)

	if !adapterReg.Has("qfree", "asc") {
		t.Fatalf("qfree-asc not registered; keys=%v", adapterReg.Keys())
	}
	conn := map[string]any{
		"snmp": map[string]any{"host": "10.0.0.5", "community": "public"},
		"api":  map[string]any{"url": "https://example/qfree"},
	}
	a, err := adapterReg.Build("qfree", "asc", "asc-100", conn)
	if err != nil {
		t.Fatalf("Build qfree-asc: %v", err)
	}
	defer a.Close()

	if _, ok := a.(adapter.SnapshotReader); !ok {
		t.Fatal("qfree-asc must implement SnapshotReader")
	}
	sc, ok := a.(interface {
		SNMPClient() internalsnmp.SNMPClient
	})
	if !ok || sc.SNMPClient() == nil {
		t.Error("qfree-asc must expose a non-nil SNMPClient() for the command path")
	}
	// Read is network-free with the stub translator (returns an empty Reading).
	if _, err := a.(adapter.SnapshotReader).Read(context.Background()); err != nil {
		t.Fatalf("Read with stub translator: %v", err)
	}
}

// A connection without an snmp sub-block is rejected.
func TestRegisterTo_RequiresSnmpBlock(t *testing.T) {
	adapterReg := adapter.NewRegistry()
	transReg := translator.NewTranslatorRegistry()
	translator.RegisterStubs(transReg)
	RegisterTo(adapterReg, transReg)

	if _, err := adapterReg.Build("qfree", "asc", "asc-100", map[string]any{"api": map[string]any{}}); err == nil {
		t.Fatal("expected error when connection.snmp is missing")
	}
}
```
